def threeSum(nums):
    nums.sort()
    res = []
    n = len(nums)
    for i in range(n):
        if nums[i] > 0:
            break
        if i > 0 and nums[i] == nums[i-1]:
            continue
        L = i + 1
        R = n - 1
        while L < R:
            s = nums[i] + nums[L] + nums[R]
            if s == 0:
                res.append([nums[i], nums[L], nums[R]])
                while L < R and nums[L] == nums[L+1]:
                    L += 1
                while L < R and nums[R] == nums[R-1]:
                    R -= 1
                L += 1
                R -= 1
            elif s < 0:
                L += 1
            else:
                R -= 1
    return res

if __name__ == "__main__":
    arr = [-1,0,1,2,-1,-4]
    print(f"输入数组：{arr}")
    print(f"输出：{threeSum(arr)}")
